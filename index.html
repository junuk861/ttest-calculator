function parseInput(str){

  return str.split(',').map(x=>parseFloat(x.trim())).filter(x=>!isNaN(x));

}


function mean(a){ return a.reduce((x,y)=>x+y,0)/a.length; }


function variance(a,m){

  return a.reduce((x,y)=>x+(y-m)**2,0)/(a.length-1);

}


function sem(a){

  return Math.sqrt(variance(a,mean(a))/a.length);

}


function runTest(){


  let g1 = parseInput(document.getElementById("g1").value);

  let g2 = parseInput(document.getElementById("g2").value);


  let type = document.getElementById("testType").value;

  let tail = document.getElementById("tail").value;


  let m1 = mean(g1), m2 = mean(g2);

  let v1 = variance(g1,m1), v2 = variance(g2,m2);


  let n1 = g1.length, n2 = g2.length;


  let t, df;


  if(type === "paired"){

    let diff = g1.map((x,i)=>x - g2[i]);

    let md = mean(diff);

    let vd = variance(diff, md);


    t = md / Math.sqrt(vd / diff.length);

    df = diff.length - 1;

  }


  else if(type === "student"){

    let pooled = ((n1-1)*v1 + (n2-1)*v2)/(n1+n2-2);

    t = (m1-m2)/Math.sqrt(pooled*(1/n1 + 1/n2));

    df = n1+n2-2;

  }


  else{ // Welch

    t = (m1-m2)/Math.sqrt(v1/n1 + v2/n2);

    df = Math.pow(v1/n1 + v2/n2,2) /

         ((Math.pow(v1/n1,2)/(n1-1)) + (Math.pow(v2/n2,2)/(n2-1)));

  }


  let p;


  if(tail === "two"){

    p = 2*(1 - jStat.studentt.cdf(Math.abs(t), df));

  } else {

    p = 1 - jStat.studentt.cdf(Math.abs(t), df);

  }


  document.getElementById("result").innerHTML =

    `Mean1 = ${m1.toFixed(3)} ± ${sem(g1).toFixed(3)}<br>

     Mean2 = ${m2.toFixed(3)} ± ${sem(g2).toFixed(3)}<br>

     t = ${t.toFixed(4)}<br>

     df = ${df.toFixed(2)}<br>

     p = ${p.toExponential(3)}`;


  drawPlot(g1,g2);

}


function drawPlot(g1,g2){


  let ctx = document.getElementById('plot').getContext('2d');


  if(window.chart) window.chart.destroy();


  window.chart = new Chart(ctx,{

    type: 'scatter',

    data:{

      datasets:[

        {

          label:'Group1',

          data: g1.map((y)=>({x:1,y:y}))

        },

        {

          label:'Group2',

          data: g2.map((y)=>({x:2,y:y}))

        }

      ]

    },

    options:{

      scales:{

        x:{ticks:{callback:(v)=>v==1?'G1':'G2'}}

      }

    }

  });

}


// CSV 업로드

document.getElementById("csvFile").addEventListener("change", function(e){

  let file = e.target.files[0];

  let reader = new FileReader();


  reader.onload = function(e){

    let rows = e.target.result.split("\n");

    let g1 = [], g2 = [];


    rows.forEach(r=>{

      let cols = r.split(",");

      if(cols.length>=2){

        g1.push(parseFloat(cols[0]));

        g2.push(parseFloat(cols[1]));

      }

    });


    document.getElementById("g1").value = g1.join(",");

    document.getElementById("g2").value = g2.join(",");

  };


  reader.readAsText(file);

});
